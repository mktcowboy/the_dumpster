    Schema

    companies (
  ticker TEXT PRIMARY KEY, name, exchange,
  price, price_change, market_cap_millions,
  sector, industry, ceo, employees, address,
  ipo_date, cik, isin, cusip, website,
  description, business,
  source_overview, source_financials, captured_at
)

facts (
  ticker, dataset, period, period_type,
  as_of, section, item, value, unit,
  PRIMARY KEY (ticker, dataset, period, section, item)
)

news (
  ticker, published, source, title, url,
  PRIMARY KEY (ticker, url)
)

    dataset is one of overview, income, balance, cashflow, capital_structure, working_capital, growth, or quarterly.

    Example queries

    -- Revenue and net income by year
SELECT period,
       MAX(CASE WHEN dataset='income'
                AND item='Sales/Revenue/Turnover'
                THEN value END) AS revenue,
       MAX(CASE WHEN dataset='income'
                AND item='Net Income, GAAP'
                THEN value END) AS net_income
FROM facts
GROUP BY period
ORDER BY period;

-- Full 2025 balance sheet, in order
SELECT section, item, value, unit
FROM facts
WHERE dataset='balance' AND period='2025'
ORDER BY rowid;

-- Latest quarterly revenue
SELECT period, value
FROM facts
WHERE dataset='quarterly' AND item='Revenue'
  AND period_type='quarter'
ORDER BY period;

-- Headlines
SELECT published, source, title
FROM news ORDER BY published DESC;
